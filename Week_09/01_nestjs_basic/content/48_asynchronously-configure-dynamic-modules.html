ตั้งค่า Dynamic Modules แบบ Asynchronous

เมื่อจบบทนี้ คุณจะสามารถ

  • อธิบายปัญหาลำดับการโหลด: ทำไม register(options) ใน imports array อาจอ่าน process.env ก่อนที่ ConfigModule จะเติมค่าให้
  • เพิ่มเมธอด registerAsync() ที่รับ useFactory + inject ให้ DatabaseModule ของบทที่ 39
  • รู้ว่าแพทเทิร์น forRootAsync/registerAsync คือสิ่งเดียวกับที่ dynamic module ทั่ว ecosystem ของ Nest ใช้

ปัญหา: ลำดับการประเมินค่าใน imports array

ค่าใน imports array ถูกประเมินตอนที่ JavaScript โหลดไฟล์ module — ก่อนที่ Nest จะเริ่ม bootstrap เสียอีก นั่นแปลว่า DatabaseModule.register({ connectionString: process.env.DATABASE_URL }) จากบทที่ 39 อ่านค่า env ทันที ณ จุดประกาศ ถ้าลำดับการโหลดไฟล์ทำให้ ConfigModule ยังไม่ได้อ่าน .env เข้า process.env ค่าที่ PrismaPg ได้ก็จะเป็น undefined แล้วการต่อฐานข้อมูลล้มเหลวแบบงง ๆ ปัญหาเดียวกันเกิดกับ forRoot(options) ของ library module ทุกตัวที่รับ options เป็น object ตรง ๆ

ทางแก้คือเลื่อนการสร้าง options object ออกไปไว้ใน factory function ซึ่ง Nest จะเรียกระหว่าง dependency injection phase — หลังจาก ConfigModule ทำงานเสร็จแล้ว และดีกว่านั้นคือ factory สามารถ inject ConfigService มาอ่านค่าที่ผ่านการ validate จากบทที่ 44 แล้วได้เลย

แพทเทิร์นของ ecosystem: forRootAsync / registerAsync

dynamic module ที่ดีของ Nest จะมีเมธอดคู่กันเสมอ: forRoot(options) สำหรับค่าคงที่ และ forRootAsync({ useFactory, inject }) สำหรับค่าที่ต้องรอ provider อื่น ตัวอย่างจาก MongooseModule ที่เราจะเจอในบทที่ 72 (และ JwtModule, CacheModule ฯลฯ ก็หน้าตาเดียวกัน)

// รูปแบบที่พบทั่ว ecosystem — ตัวอย่างจาก @nestjs/mongoose
MongooseModule.forRootAsync({
  inject: [ConfigService],
  useFactory: (configService: ConfigService) => ({
    uri: configService.get<string>('MONGO_URI'),
  }),
}),

โค้ดข้างในเหมือน forRoot ทุกบรรทัด ต่างกันแค่ถูกห่อด้วย function ทำให้ค่าถูกอ่าน ตอนถูกเรียก ไม่ใช่ ตอนไฟล์ถูกโหลด — เท่านี้ปัญหาลำดับก็หายไป และ inject ก็คือ inject ตัวเดียวกับ factory provider ในบทที่ 37

สร้าง registerAsync ให้ DatabaseModule ของเราเอง

เบื้องหลัง forRootAsync ไม่มีอะไรมากกว่า factory provider ที่เรารู้จักแล้ว: แทนที่จะ useValue client ที่สร้างเสร็จ เราสร้าง provider 'DATABASE_OPTIONS' จาก factory ของผู้เรียก แล้วให้ provider 'CONNECTION' inject options นั้นต่ออีกที

// database.module.ts
import { DynamicModule, Module, ModuleMetadata } from '@nestjs/common';
import { PrismaPg } from '@prisma/adapter-pg';
import { PrismaClient } from '../generated/prisma/client.js';

export interface DatabaseOptions {
  connectionString: string;
}

export interface DatabaseAsyncOptions extends Pick<ModuleMetadata, 'imports'> {
  useFactory: (...args: any[]) => DatabaseOptions | Promise<DatabaseOptions>;
  inject?: any[];
}

@Module({})
export class DatabaseModule {
  static register(options: DatabaseOptions): DynamicModule {
    return {
      module: DatabaseModule,
      providers: [
        {
          provide: 'CONNECTION',
          useValue: new PrismaClient({ adapter: new PrismaPg(options) }),
        },
      ],
      exports: ['CONNECTION'],
    };
  }

  static registerAsync(options: DatabaseAsyncOptions): DynamicModule {
    return {
      module: DatabaseModule,
      imports: options.imports ?? [],
      providers: [
        {
          provide: 'DATABASE_OPTIONS',
          useFactory: options.useFactory, // 👈 ถูกเรียกใน DI phase ไม่ใช่ตอนโหลดไฟล์
          inject: options.inject ?? [],
        },
        {
          provide: 'CONNECTION',
          useFactory: (dbOptions: DatabaseOptions) =>
            new PrismaClient({ adapter: new PrismaPg(dbOptions) }),
          inject: ['DATABASE_OPTIONS'],
        },
      ],
      exports: ['CONNECTION'],
    };
  }
}

ฝั่งผู้ใช้เปลี่ยนจาก register เป็น registerAsync แล้วส่ง ConfigService เข้ามาผ่าน inject — และต้องส่ง imports: [ConfigModule] มาด้วย เพราะ provider DATABASE_OPTIONS ถูกสร้างอยู่ ภายใน DatabaseModule ซึ่งมองไม่เห็น ConfigService ของ AppModule (ConfigModule.forRoot() ไม่ได้เป็น global โดย default) นี่คือหน้าที่ของ imports ใน DatabaseAsyncOptions ที่เรา Pick มาจาก ModuleMetadata

// coffee-rating.module.ts
import { ConfigModule, ConfigService } from '@nestjs/config';

imports: [
  DatabaseModule.registerAsync({
    imports: [ConfigModule], // 👈 ให้ DatabaseModule มองเห็น ConfigService
    inject: [ConfigService],
    useFactory: (configService: ConfigService) => ({
      connectionString: configService.getOrThrow<string>('DATABASE_URL'),
    }),
  }),
]

เราอ่านค่าผ่าน configService.getOrThrow(...) แทนการแตะ process.env ตรง ๆ ซึ่งทำให้ options ผ่านการ validate จาก schema ในบทที่ 44 ด้วย — และต่างจาก get() ที่คืน string | undefined เมธอด getOrThrow() คืน string ล้วน ๆ (throw ทันทีถ้า key ไม่มี) จึงส่งเข้า connectionString: string ได้โดยไม่ต้องใช้ ! แบบบทที่ 39 อีก ถ้า ConfigModule ไม่ได้ประกาศ isGlobal: true ให้ส่ง imports: [ConfigModule] เข้า registerAsync ด้วย — นั่นคือเหตุผลที่ interface รับ imports

register(options)registerAsync({ useFactory, inject })
เวลาอ่านค่า optionsตอนไฟล์ถูก import (เร็วเกินไปสำหรับ env)ตอน DI phase หลัง ConfigModule พร้อม

เหมาะกับค่าคงที่ที่รู้ตั้งแต่เขียนโค้ดค่าจาก environment หรือ service อื่น

PrismaService ตัวหลักของแอปไม่ต้องใช้แพทเทิร์นนี้ เพราะมันอ่าน process.env.DATABASE_URL ใน constructor ซึ่งรันใน DI phase อยู่แล้ว (หลัง ConfigModule โหลด .env) — ปัญหาลำดับเกิดเฉพาะกับค่าที่ถูกประเมินที่ระดับ top-level ของ imports array เท่านั้น

ย้าย DatabaseModule ไปใช้ registerAsync

เพิ่ม registerAsync() ให้ DatabaseModule ตามโค้ดด้านบน แล้วใน CoffeeRatingModule เปลี่ยน DatabaseModule.register({...}) เป็น DatabaseModule.registerAsync({ inject: [ConfigService], useFactory }) จากนั้น restart ให้แน่ใจว่า 'CONNECTION' ยัง query ฐานข้อมูลได้

// เปลี่ยนจาก DatabaseModule.register({ connectionString: process.env.DATABASE_URL }) // เป็น DatabaseModule.registerAsync({ imports: [ConfigModule], inject: [ConfigService], useFactory: (config: ConfigService) => ({ connectionString: config.getOrThrow<string>('DATABASE_URL'), }), }) npm run start:dev

ผลลัพธ์ที่คาดหวัง

แอป start และ query ผ่าน 'CONNECTION' ได้เหมือนบทที่ 39 แต่ตอนนี้การอ่าน connection string ถูกเลื่อนไปอยู่หลัง ConfigModule เสมอ และผ่าน ConfigService ที่ validate แล้ว จึงไม่ขึ้นกับลำดับ import ในไฟล์อีกต่อไป

ตรวจว่าใช้งานได้จริง

  • แอป start โดยไม่มี error และ $queryRaw ผ่าน 'CONNECTION' ยังคืนผลลัพธ์
  • ใส่ console.log('factory called') ใน useFactory — ต้องพิมพ์หลัง log ของ ConfigModule และก่อน CoffeeRatingService ถูกสร้าง
  • ลองสลับตำแหน่ง ConfigModule.forRoot() ไปไว้ท้ายสุดของ imports array ใน AppModule — แอปยังทำงานได้ เพราะ factory ถูกเรียกทีหลังเสมอ
  • ไม่มีการอ่าน process.env ที่ระดับ top-level ของ imports array เหลืออยู่ในโปรเจกต์
