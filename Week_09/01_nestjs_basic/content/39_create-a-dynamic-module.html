สร้าง Dynamic Module

เมื่อจบบทนี้ คุณจะสามารถ

  • อธิบายข้อจำกัดของ static module ที่ config ตายตัวจนใช้ซ้ำข้ามบริบทไม่ได้
  • เขียนเมธอด static register(options) ที่ return DynamicModule
  • ใช้ dynamic module รับค่าจากผู้เรียก (เช่น connection options) ตอน import

ปัญหา: module แบบ static ปรับแต่งไม่ได้

ลองนึกถึง DatabaseModule ที่สร้าง connection provider ไว้ภายใน: ถ้าเขียน host/port ตายตัวใน module ผู้ใช้ทุกคนจะได้ connection เดียวกันหมด ปรับต่อไม่ได้เลย นี่คือรูปแบบที่เราจะเริ่มจากมันก่อน

# สร้าง DatabaseModule
nest g mo database
// ความพยายามแรก: provider "CONNECTION" ที่ config ฝังตายตัว
{
  provide: 'CONNECTION',
  useValue: new PrismaClient({
    adapter: new PrismaPg({
      connectionString: 'postgresql://postgres:pass123@localhost:5432/postgres',
    }),
  }),
}

Module ประเภทนี้ต้องการสิ่งเดียวกับที่ library modules อย่าง ConfigModule.forRoot(...) (บทที่ 42) หรือ MongooseModule.forRoot(...) (บทที่ 72) ทำ: ให้ผู้ import เป็นคนส่ง config เข้ามา — กลไกเบื้องหลังของ pattern นั้นคือ Dynamic Module

ทางออก: static register() ที่ return DynamicModule

Dynamic module คือ module ที่มี static method (ตามธรรมเนียมชื่อ register, forRoot หรือ forFeature) รับ options แล้ว return object ตาม interface DynamicModule ซึ่งประกอบ providers ขึ้น ณ ตอนเรียก จึงฝังค่าจากผู้ใช้ลงไปได้

// database.module.ts
import { DynamicModule, Module } from '@nestjs/common';
import { PrismaPg } from '@prisma/adapter-pg';
import { PrismaClient } from '../generated/prisma/client.js';

export interface DatabaseOptions {
  connectionString: string;
}

// โครงเริ่มต้น
@Module({})
export class DatabaseModule {
  static register(options: DatabaseOptions): DynamicModule {  }
}

// เวอร์ชันสมบูรณ์: สร้าง CONNECTION จาก options ที่ผู้เรียกส่งมา
@Module({})
export class DatabaseModule {
  static register(options: DatabaseOptions): DynamicModule {
    return {
      module: DatabaseModule,
      providers: [
        {
          provide: 'CONNECTION',
          useValue: new PrismaClient({ adapter: new PrismaPg(options) }),
        },
      ],
      exports: ['CONNECTION'],
    };
  }
}

ฝั่งผู้ใช้ import ด้วยการเรียกเมธอดแทนการอ้างชื่อ class เฉย ๆ

// ใช้ dynamic DatabaseModule ใน imports ของ module อื่น
imports: [
  DatabaseModule.register({ // ส่งค่าจริงเข้าไปตอน import
    connectionString: process.env.DATABASE_URL!, // ! กัน strict mode ชั่วคราว — บทที่ 48 เปลี่ยนเป็น getOrThrow
  })
]

PrismaPg รับ options ชุดเดียวกับ pg.Pool (เช่น max จำนวน connection สูงสุด) interface DatabaseOptions จึงเป็นที่ที่เหมาะจะขยายให้ผู้ใช้ปรับ pool ได้ในอนาคต — นี่คือคุณค่าของการให้ผู้ import เป็นคนส่ง config

provider 'CONNECTION' ตัวนี้คือ PrismaClient ตัวที่สองของแอป (มี connection pool ของตัวเอง) เราสร้างมันเพื่อเรียนรู้กลไกของ dynamic module เท่านั้น — โค้ดจริงของ CoffeesService ยังใช้ PrismaService จาก PrismaModule ตามเดิม อย่าเผลอเปิดหลาย client ในแอปจริงโดยไม่จำเป็น

สร้าง DatabaseModule.register ของตัวเอง

generate DatabaseModule, เขียน static register(options: DatabaseOptions) ตามตัวอย่าง แล้ว import จาก CoffeeRatingModule (บทที่ 32) พร้อมส่ง connectionString จากนั้น inject token 'CONNECTION' ที่ไหนสักแห่งแล้วลอง query เพื่อพิสูจน์ว่า client ที่ได้ต่อฐานข้อมูลตาม options ที่ส่งจริง

nest g mo database // coffee-rating.module.ts imports: [ DatabaseModule.register({ connectionString: process.env.DATABASE_URL!, // ! กัน strict mode ชั่วคราว — บทที่ 48 เปลี่ยนเป็น getOrThrow }), ] // coffee-rating.service.ts — ทดลองใช้ CONNECTION constructor(@Inject('CONNECTION') private readonly connection: PrismaClient) {} async onModuleInit() { console.log(await this.connection.$queryRaw`SELECT current_database()`); } npm run start:dev

ผลลัพธ์ที่คาดหวัง

แอป start ได้ปกติ และ log แสดงผล [ { current_database: 'postgres' } ] จาก client ที่ inject มา — module เดียวกันนี้สามารถถูก import ที่อื่นด้วย connectionString คนละชุด (เช่น ชี้ไป test-db ในบทที่ 68) ได้ทันที ซึ่งเป็นสิ่งที่ static module ทำไม่ได้

ตรวจว่าใช้งานได้จริง

  • ตรวจว่า register() return object ที่มีทั้ง module: DatabaseModule, providers และ exports
  • $queryRaw ผ่าน 'CONNECTION' ได้ผลลัพธ์กลับมา แสดงว่า options ถูกส่งถึง PrismaPg จริง
  • ทดลอง import DatabaseModule.register() ด้วย connectionString ปลอมใน module อื่น — query จาก instance นั้นต้อง error ขณะที่ instance แรกยังทำงาน
  • CoffeesService ยังใช้ PrismaService ตามเดิม ไม่ได้ถูกเปลี่ยนมาใช้ 'CONNECTION'
