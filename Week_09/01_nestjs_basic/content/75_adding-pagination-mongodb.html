เพิ่ม Pagination (MongoDB)

เมื่อจบบทนี้ คุณจะสามารถ

  • สร้าง PaginationQueryDto ที่ใช้ร่วมกันได้ทั้งแอปไว้ในโฟลเดอร์ common
  • Validate ค่า limit/offset ให้เป็นจำนวนบวกและเป็น optional ด้วย class-validator
  • ต่อ query ของ Mongoose ด้วย skip() และ limit() เพื่อตัดหน้าข้อมูลจริงที่ฐานข้อมูล

DTO สำหรับ pagination ควรอยู่ที่ common ไม่ใช่ใน coffees

Pagination ไม่ใช่เรื่องเฉพาะของ coffees — resource อื่นในอนาคตก็ต้องใช้เหมือนกัน เราจึงสร้าง DTO ไว้ในโฟลเดอร์กลาง src/common/dto/ โดยใช้ Nest CLI ช่วย generate (ใส่ --no-spec เพื่อไม่สร้างไฟล์ test)

# Generate PaginationQueryDto (ไม่เอาไฟล์ test)
nest g class common/dto/pagination-query.dto --no-spec
/* PaginationQueryDto - FINAL CODE */
import { Type } from 'class-transformer';
import { IsOptional, IsPositive, Min } from 'class-validator';

export class PaginationQueryDto {
  @Type(() => Number)
  @IsOptional()
  @IsPositive()
  limit: number;

  @Type(() => Number)
  @IsOptional()
  @Min(0) // 👈 ไม่ใช่ @IsPositive() — offset=0 (เริ่มจากแถวแรก) เป็นค่าที่ถูกต้อง
  offset: number;
}

เส้นทาง MongoDB นี้แตกมาจากจุดจบของบทที่ 18 ซึ่งเปิดแค่ whitelist + forbidNonWhitelisted + transform — ไม่มี enableImplicitConversion (ตัวนั้นเพิ่มในบทที่ 28 ของเส้นทาง PostgreSQL/Prisma ซึ่งเส้นทางนี้ไม่ได้ผ่าน) query string อย่าง "10" จึงยังเป็น string ตอนเข้า validator ถ้าไม่ติด @Type(() => Number) ให้แต่ละ field เอง @IsPositive()/@Min() จะ reject ทุกคำขอที่ส่ง query มา (เพราะ "10" ไม่ใช่ตัวเลขในสายตาของ validator) — โค้ดด้านบนจึงติด @Type() ไว้ที่ตัว DTO เองแทน

ส่งต่อให้ Mongoose ด้วย skip/limit

ใน controller ให้ findAll() รับ @Query() paginationQuery: PaginationQueryDto แล้วส่งต่อมาที่ service ซึ่งนำค่าไปต่อท้าย query: skip(offset) ข้ามจำนวน document ตามที่ระบุ และ limit(limit) จำกัดจำนวนที่ตอบกลับ ทั้งหมดทำงานที่ฐานข้อมูล ไม่ใช่ดึงมาทั้งก้อนแล้วค่อยตัดในแอป

/* CoffeesService - FINAL CODE */
import { Injectable, NotFoundException } from '@nestjs/common';
import { InjectModel } from '@nestjs/mongoose';
import { Model } from 'mongoose';
import { PaginationQueryDto } from '../common/dto/pagination-query.dto.js';
import { CreateCoffeeDto } from './dto/create-coffee.dto.js';
import { UpdateCoffeeDto } from './dto/update-coffee.dto.js';
import { Coffee } from './entities/coffee.entity.js';

@Injectable()
export class CoffeesService {
  constructor(
    @InjectModel(Coffee.name) private readonly coffeeModel: Model<Coffee>,
  ) {}

  findAll(paginationQuery: PaginationQueryDto) {
    const { limit, offset } = paginationQuery;
    return this.coffeeModel
      .find()
      .skip(offset)
      .limit(limit)
      .exec();
  }

  async findOne(id: string) {
    const coffee = await this.coffeeModel.findOne({ _id: id }).exec();
    if (!coffee) {
      throw new NotFoundException(`Coffee #${id} not found`);
    }
    return coffee;
  }

  create(createCoffeeDto: CreateCoffeeDto) {
    const coffee = new this.coffeeModel(createCoffeeDto);
    return coffee.save();
  }

  async update(id: string, updateCoffeeDto: UpdateCoffeeDto) {
    const existingCoffee = await this.coffeeModel
      .findOneAndUpdate({ _id: id }, { $set: updateCoffeeDto }, { new: true })
      .exec();

    if (!existingCoffee) {
      throw new NotFoundException(`Coffee #${id} not found`);
    }
    return existingCoffee;
  }

  async remove(id: string) {
    const coffee = await this.findOne(id);
    return coffee.deleteOne();
  }
}

แบ่งหน้าข้อมูลกาแฟ

สร้าง DTO, อัปเดต findAll ทั้งใน controller และ service ตามด้านบน แล้ว seed กาแฟสัก 5 รายการผ่าน POST ก่อนทดลองยิงแบบมีและไม่มี query parameters

GET http://localhost:3000/coffees GET http://localhost:3000/coffees?limit=2 GET http://localhost:3000/coffees?limit=2&offset=2 GET http://localhost:3000/coffees?limit=-5 # ควรถูก validator ปฏิเสธ

ผลลัพธ์ที่คาดหวัง

ไม่ใส่ query ได้ครบทุกรายการเหมือนเดิม (Mongoose ข้าม skip/limit ที่เป็น undefined), limit=2 ได้ 2 รายการแรก, เพิ่ม offset=2 ได้รายการที่ 3–4 และค่าติดลบได้ 400 Bad Request พร้อมข้อความจาก class-validator

ตรวจว่าใช้งานได้จริง

  • ผลของ limit=2 กับ limit=2&offset=2 ไม่มีรายการซ้ำกัน
  • limit=0 หรือ limit=-5 ตอบ 400 ไม่ใช่ข้อมูลทั้งหมด
  • PaginationQueryDto อยู่ใน src/common/dto/ และไม่ import อะไรจากโฟลเดอร์ coffees
  • ยิงโดยไม่ใส่ query เลยแล้วยังได้ 200 พร้อมรายการครบ — ยืนยันว่า optional จริง
