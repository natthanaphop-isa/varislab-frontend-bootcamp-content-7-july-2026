สร้าง e2e Test แรก

เมื่อจบบทนี้ คุณจะสามารถ

  • ทดสอบ feature module เดี่ยว ๆ (Coffees) โดย import เฉพาะ module ที่ต้องการเข้า TestingModule
  • ตั้ง test database แยกจาก database พัฒนาใน docker-compose.yml และสตาร์ต/ปิดอัตโนมัติด้วย pre/post hooks ของ npm
  • วางโครง test ล่วงหน้าด้วย it.todo ก่อนเติม logic จริง

ข้อดีของ modular architecture ตอนทดสอบ

เพราะเราห่อ feature ไว้ใน modules มาตลอดคอร์ส ตอนนี้ได้ผลตอบแทน: จะทดสอบ feature ไหน ก็ import เฉพาะ module นั้นเข้า TestingModule ไม่ต้องลาก AppModule ทั้งใบ ในบทนี้เราจะทดสอบ CRUD endpoints ของ CoffeesModule ที่สร้างกันมาทั้งคอร์ส

แต่ e2e ต้องมี database จริง — และไม่ควรใช้ตัวเดียวกับที่เราพัฒนา เพราะ test จะเขียน/ลบข้อมูลทิ้ง วิธีของคอร์สคือเพิ่ม service ที่สองใน docker-compose.yml เป็น test-db ที่ map คนละ port

# docker-compose.yml - FINAL CODE
version: "3"

services:
  db:
    image: postgres
    restart: always
    ports:
      - "5432:5432"
    environment:
      POSTGRES_PASSWORD: pass123
  test-db:
    image: postgres
    restart: always
    ports:
      - "5433:5432" # 👈 ใช้ 5433 เพราะ 5432 เป็นของ db ปกติ
    environment:
      POSTGRES_PASSWORD: pass123

ให้ npm สตาร์ตและเก็บกวาด database ให้อัตโนมัติ

npm รู้จัก hook ชื่อ pre<script> และ post<script>: ถ้ามี script ชื่อ pretest:e2e มันจะรันก่อน test:e2e เสมอ และ posttest:e2e รันหลังจบ เราใช้คู่นี้ยก test-db ขึ้นก่อนรัน และ tear down ทิ้งเมื่อเสร็จ

// package.json
"pretest:e2e": "docker-compose up -d test-db && sleep 3 && DATABASE_URL=postgresql://postgres:pass123@localhost:5433/postgres npx prisma migrate deploy",
"test:e2e": "vitest run --config ./vitest.config.e2e.ts",
"posttest:e2e": "docker-compose stop test-db && docker-compose rm -f test-db"

script test:e2e ตรงกลางคือตัวเดิมที่ Nest เตรียมให้ ไม่ต้องแก้ — npm จะรัน pretest:e2e ก่อนและ posttest:e2e หลังให้เองตามชื่อ

test-db ที่เพิ่งถูกสร้างเป็นฐานข้อมูลเปล่า จึงต้องสร้าง schema ให้ก่อน: sleep 3 รอให้ Postgres พร้อมรับ connection แล้ว prisma migrate deploy apply migration ทั้งหมดจากบทที่ 31 ลงไป โดยส่ง DATABASE_URL ของ test-db เป็น environment variable หน้าคำสั่ง — ค่านี้ชนะค่าใน .env เพราะ dotenv ไม่เขียนทับตัวแปรที่มีอยู่แล้ว

ถ้าคุณข้ามบทที่ 31 และยังใช้ db push อยู่ ให้เปลี่ยนคำสั่งท้ายเป็น npx prisma db push แทน ส่วนผู้ใช้ Windows ให้ติดตั้ง cross-env (npm i -D cross-env) แล้วนำหน้าคำสั่งด้วย cross-env DATABASE_URL=... เพราะ syntax ตั้งตัวแปรหน้าคำสั่งแบบนี้ใช้ได้เฉพาะ shell ตระกูล Unix

เมื่อ script ข้างใน error (เช่น Vitest fail) post hook อาจไม่รัน! ทางเลือกคือ (1) รัน posttest:e2e เองเมื่อ Vitest พัง หรือ (2) ใช้ไลบรารี npm-run-all (npm i -D npm-run-all) กับ flag --continue-on-error โดยย้ายขั้น teardown มาเป็น script ปกติ เช่น "test:e2e": "npm-run-all test:e2e:run test:e2e:teardown --continue-on-error" เพื่อให้เก็บกวาดเสมอแม้ test จะแดง

โครง test: ประกาศก่อน เติมทีหลัง

/* test/coffees/coffees.e2e-spec.ts - FINAL CODE */
import { INestApplication } from '@nestjs/common';
import { TestingModule, Test } from '@nestjs/testing';
import { CoffeesModule } from '../../src/coffees/coffees.module.js';
import { PrismaModule } from '../../src/prisma/prisma.module.js';

// 👈 ชี้ PrismaService ไป test-db (port 5433) ก่อนที่ module ใด ๆ จะถูกสร้าง
process.env.DATABASE_URL =
  'postgresql://postgres:pass123@localhost:5433/postgres';

describe('[Feature] Coffees - /coffees', () => {
  let app: INestApplication;

  beforeAll(async () => {
    const moduleFixture: TestingModule = await Test.createTestingModule({
      imports: [
        CoffeesModule, // 👈 เฉพาะ module ที่ทดสอบ
        PrismaModule,  // 👈 ให้ PrismaService แก่ CoffeesModule (ไม่ต้องลาก AppModule)
      ],
    }).compile();

    app = moduleFixture.createNestApplication();
    await app.init();
  });

  it.todo('Create [POST /]');
  it.todo('Get all [GET /]');
  it.todo('Get one [GET /:id]');
  it.todo('Update one [PATCH /:id]');
  it.todo('Delete one [DELETE /:id]');

  afterAll(async () => {
    await app.close();
  });
});

เราตั้ง process.env.DATABASE_URL ที่บรรทัดบนสุดของไฟล์ เพราะ PrismaService อ่านค่านี้ใน constructor — ต้องมีค่าก่อน compile() สร้าง provider และเพราะ test ไม่ได้ import ConfigModule จึงไม่มีใครโหลด .env มาทับค่านี้

it.todo ประกาศรายการ test ที่ตั้งใจจะเขียนโดยยังไม่มี body — Vitest จะรายงานเป็น todo แยกจาก pass/fail ทำให้เราวางแผนครอบคลุม CRUD ทั้ง 5 เส้นได้ก่อนแล้วค่อยเติม logic ในบทถัดไป

ตั้ง test database และโครง suite ของ Coffees

เพิ่ม test-db ใน compose file, เพิ่ม pre/post hooks ใน package.json แล้วสร้าง test/coffees/coffees.e2e-spec.ts ตามโค้ดด้านบน

# รันเฉพาะ e2e ของ feature coffees npm run test:e2e -- coffees # สังเกต: pretest:e2e ยก test-db ขึ้นเอง และ posttest:e2e เก็บกวาดให้

ผลลัพธ์ที่คาดหวัง

Suite [Feature] Coffees - /coffees boot สำเร็จโดยต่อกับ Postgres ที่ port 5433 และ Vitest สรุปผลเป็น 5 todo tests โดยไม่มี fail — พร้อมสำหรับเติม logic ในบทถัดไป

ตรวจว่าใช้งานได้จริง

  • ระหว่าง test รัน ใช้ docker ps ดูว่า container ของ test-db ขึ้นที่ port 5433 และหายไปหลัง test จบ
  • log ของ pretest:e2e แสดง prisma migrate deploy apply migrations ลง test-db สำเร็จ
  • ผลรายงาน Vitest แสดง 5 todo ใน summary
  • ยืนยันว่า database พัฒนา (port 5432) ไม่ถูก test แตะ: ข้อมูล coffees เดิมยังอยู่ครบ
  • ลองทำให้ Vitest error (เช่นพิมพ์ syntax ผิดชั่วคราว) แล้วสังเกตว่า post hook รันหรือไม่ — จะได้เข้าใจกับดักใน warning ด้านบนจากของจริง
