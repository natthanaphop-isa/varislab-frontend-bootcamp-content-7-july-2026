บทเรียน 07 · เริ่มคิดแบบ JavaScript (Thinking in JavaScript)

ตรวจสอบข้อมูลด้วย if/else

if/else Validation

เมื่อจบบทนี้ คุณจะสามารถ

  • ตัดช่องว่างหน้าหลังข้อความด้วย .trim()
  • เขียนเงื่อนไขด้วย if, === และ || เพื่อเช็กว่าช่องสำคัญว่างหรือไม่
  • ใช้ return หยุดการทำงานของ function เมื่อข้อมูลไม่ครบ
  • ใช้ || กำหนดข้อความสำรองเมื่อค่าว่าง และได้ script.js ฉบับสมบูรณ์

นี่คือส่วนที่สำคัญที่สุดของโปรเจกต์ เพราะเราเริ่มเขียน logic จริง ๆ แล้ว ในเว็บจริงเราไม่ควรปล่อยให้ผู้ใช้กดส่งฟอร์มทั้งที่ยังไม่กรอกข้อมูลสำคัญ เราจึงต้องตรวจสอบเงื่อนไขก่อน — ขั้นตอนนี้เรียกว่า validation

Step 6: script.js ฉบับสมบูรณ์

แก้ script.js ให้เป็นแบบนี้ แล้วเราจะไล่อธิบายทีละส่วน:

const inquiryForm = document.getElementById("inquiryForm");
const fullNameInput = document.getElementById("fullName");
const goalInput = document.getElementById("goal");
const levelSelect = document.getElementById("level");
const messageInput = document.getElementById("message");
const resultBox = document.getElementById("result");

inquiryForm.addEventListener("submit", function (event) {
  event.preventDefault();

  const fullName = fullNameInput.value.trim();
  const goal = goalInput.value.trim();
  const level = levelSelect.value;
  const message = messageInput.value.trim();

  if (fullName === "" || goal === "" || level === "") {
    resultBox.textContent =
      "กรุณากรอกชื่อ เป้าหมาย และเลือกระดับพื้นฐานให้ครบก่อนส่งข้อมูล";
    return;
  }

  resultBox.textContent = `สวัสดีครับคุณ ${fullName}
เป้าหมายของคุณคือ ${goal}
ระดับพื้นฐานตอนนี้คือ ${level}
ข้อความเพิ่มเติม: ${message || "ไม่มีข้อความเพิ่มเติม"}`;
});

เทียบกับบทที่แล้ว มีของใหม่ 4 อย่าง: .trim(), บล็อก if, return และ ${message || "…"}

.trim() — ตัดช่องว่างหน้าหลัง

const fullName = fullNameInput.value.trim();

.trim() ตัดช่องว่าง (space, tab, ขึ้นบรรทัดใหม่) ที่อยู่หน้าและหลังข้อความออก เช่น " Man " จะกลายเป็น "Man"

เหตุผลสำคัญ: ถ้าผู้ใช้กด spacebar หลายครั้งแต่ไม่ได้พิมพ์อะไรจริง ๆ ค่าที่ได้คือ " " ซึ่งไม่ใช่ข้อความว่างในสายตา JavaScript หลัง .trim() มันจะกลายเป็น "" และถูกจับได้ว่าว่างจริง

สังเกตว่า level ไม่ได้ใส่ .trim() เพราะค่ามาจาก value ของ option ที่เรากำหนดเองอยู่แล้ว ผู้ใช้พิมพ์ช่องว่างเข้าไปไม่ได้

if — "ถ้า"

if (fullName === "" || goal === "" || level === "") {
  resultBox.textContent =
    "กรุณากรอกชื่อ เป้าหมาย และเลือกระดับพื้นฐานให้ครบก่อนส่งข้อมูล";
  return;
}

if แปลว่า "ถ้า" — โค้ดในวงเล็บปีกกา { } จะทำงานเฉพาะเมื่อเงื่อนไขในวงเล็บ ( ) เป็นจริง

  • === — เปรียบเทียบว่า "เท่ากันหรือไม่" fullName === "" คือถาม "ชื่อเป็นข้อความว่างใช่ไหม"
  • || — แปลว่า "หรือ" ถ้ามีข้อใดข้อหนึ่งเป็นจริง เงื่อนไขทั้งหมดก็เป็นจริง

แปลเงื่อนไขนี้เป็นภาษาคนได้ว่า:

ถ้า fullName ว่าง หรือ goal ว่าง หรือ level ว่าง (ยังไม่ได้เลือกระดับพื้นฐาน) → ให้แสดงข้อความเตือนผู้ใช้ แล้วหยุดการทำงาน

จำได้ไหมว่าในบทที่ 3 เราตั้ง <option value=""> ให้ตัวเลือกแรก — นี่แหละคือที่มาของ level === ""

ใช้ === (เท่ากับสามตัว) สำหรับเปรียบเทียบ ส่วน = ตัวเดียวคือการกำหนดค่าให้ตัวแปร ถ้าเขียน if (fullName = "") จะไม่ใช่การเช็กแต่เป็นการพยายามเปลี่ยนค่า และกับตัวแปร const จะเกิด error

return — หยุดตรงนี้

return ในที่นี้แปลว่า หยุดการทำงานของ function ตรงนี้เลย ถ้าข้อมูลยังไม่ครบ เราไม่ต้องการให้ JavaScript ทำงานต่อไปถึงส่วนแสดงผลลัพธ์ด้านล่าง ซึ่งจะไปเขียนทับข้อความเตือน

ถ้าลืม return โค้ดจะเขียนข้อความเตือนก่อน แล้ววิ่งต่อไปเขียนข้อความสรุปทับทันที ผู้ใช้จะไม่เคยเห็นข้อความเตือนเลย

แล้ว else อยู่ไหน?

เราเขียนแบบ if + else ตรง ๆ ก็ได้ผลเหมือนกัน:

if (fullName === "" || goal === "" || level === "") {
  resultBox.textContent =
    "กรุณากรอกชื่อ เป้าหมาย และเลือกระดับพื้นฐานให้ครบก่อนส่งข้อมูล";
} else {
  resultBox.textContent = `สวัสดีครับคุณ ${fullName} …`;
}

แต่การใช้ if + return ทำให้ส่วน else ไม่จำเป็น เพราะถ้าโค้ดวิ่งมาถึงบรรทัดหลัง if ได้ แปลว่าข้อมูลผ่านการตรวจแล้วแน่นอน เทคนิค "เช็กกรณีผิดก่อนแล้วออกเลย" นี้ทำให้โค้ดไม่ต้องซ้อนวงเล็บหลายชั้น

${message || "ไม่มีข้อความเพิ่มเติม"} — ข้อความสำรอง

ช่องข้อความเพิ่มเติมเป็นช่องไม่บังคับ จึงไม่อยู่ในเงื่อนไข if แต่ถ้าผู้ใช้ไม่ได้กรอก ผลลัพธ์จะเป็น "ข้อความเพิ่มเติม: " แล้วว่างเปล่า ดูไม่ดี เราจึงเขียน:

${message || "ไม่มีข้อความเพิ่มเติม"}

ตรงนี้ || ทำหน้าที่ต่างจากใน if เล็กน้อย: มันแปลว่า "ใช้ค่าทางซ้าย ถ้าค่าทางซ้ายว่าง ให้ใช้ค่าทางขวาแทน"

  • ถ้า message เป็น "อยากทำ Portfolio" → แสดง อยากทำ Portfolio
  • ถ้า message เป็น "" → แสดง ไม่มีข้อความเพิ่มเติม

ที่ทำงานแบบนี้ได้เพราะ JavaScript มองข้อความว่าง "" ว่าเป็น "เท็จ" (falsy) เมื่อใช้กับ || ตอนนี้แค่จำไว้ว่า ค่า || ค่าสำรอง คือวิธีสั้น ๆ ในการใส่ค่าเริ่มต้นให้ค่าที่อาจว่าง

สิ่งที่ทำผลที่ควรเห็น
กดส่งโดยไม่กรอกอะไรเลยข้อความเตือน "กรุณากรอกชื่อ เป้าหมาย และเลือกระดับพื้นฐานให้ครบก่อนส่งข้อมูล"
พิมพ์แต่ spacebar ในช่องชื่อ กรอกช่องอื่นครบข้อความเตือน (เพราะ .trim())
กรอกชื่อและเป้าหมาย แต่ไม่เลือกระดับพื้นฐานข้อความเตือน
กรอกครบ 3 ช่องสำคัญ ไม่กรอกข้อความเพิ่มเติมข้อความสรุป ลงท้ายด้วย "ข้อความเพิ่มเติม: ไม่มีข้อความเพิ่มเติม"
กรอกครบทุกช่องข้อความสรุปพร้อมข้อความเพิ่มเติมที่พิมพ์ไว้

ตอนนี้ index.html, style.css และ script.js ของคุณเสร็จสมบูรณ์แล้ว — flow ครบ: รับข้อมูล → ตรวจสอบ → ประมวลผล → แสดงผล

  • ทดสอบครบทั้ง 5 กรณีในตารางและได้ผลตรงตามที่ควรเป็น
  • อธิบายได้ว่าทำไมต้องใส่ .trim()
  • อ่านเงื่อนไข if ออกมาเป็นภาษาคนได้ และบอกได้ว่า || แปลว่าอะไร
  • บอกได้ว่าถ้าลบ return ออกจะเกิดอะไรขึ้น
