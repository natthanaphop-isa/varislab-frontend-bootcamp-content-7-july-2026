JavaScript แบบ Async

Async JavaScript

Async JavaScript

ตลอดมาที่คุณเขียน JavaScript ทุกบรรทัดที่คุณเขียนทำงานเสร็จทันที คุณบวกเลข คุณ map array คุณอ่านค่าจาก object คุณเปลี่ยนข้อความใน DOM — พอบรรทัดนั้นจบ ผลลัพธ์ก็อยู่ในมือคุณแล้ว บรรทัดถัดไปจึงหยิบไปใช้ต่อได้เลยโดยไม่ต้องคิดอะไรมาก

การขอข้อมูลจากเซิร์ฟเวอร์ไม่เป็นแบบนั้น มันใช้เวลา อาจ 50 มิลลิวินาทีถ้าเซิร์ฟเวอร์อยู่ในเครื่องเดียวกับคุณ อาจสองวินาทีถ้าเน็ตช้า หรืออาจไม่กลับมาเลยถ้าเซิร์ฟเวอร์ล่ม และความจริงข้อนี้เปลี่ยนวิธีที่เราต้องเขียนโค้ดไปทั้งหมด บทนี้คือการทำความรู้จักเครื่องมือที่ JavaScript ให้มาสำหรับจัดการกับ "ผลลัพธ์ที่ยังมาไม่ถึง" ก่อนที่เราจะเอามันไปยิง API จริงในบทที่ 12 และเอาไปใส่ใน React ในบทที่ 13

บทนี้เป็น JavaScript ล้วน ๆ ไม่มี React เลยสักบรรทัด ทุกตัวอย่างในบทนี้ให้คุณพิมพ์ลงใน console ของเบราว์เซอร์ เปิดแอปที่ http://localhost:5173 กด DevTools แล้วไปที่แท็บ Console — ที่เดียวกับที่คุณใช้มาแล้วในบทเรียนเรื่อง DevTools ของคอร์ส JavaScript พื้นฐาน พิมพ์แล้วกด Enter ดูผลจริงทุกอัน อย่าแค่อ่านผ่าน เรื่อง async เป็นเรื่องที่ "เข้าใจตอนอ่าน" กับ "เข้าใจตอนเห็นลำดับผลลัพธ์จริง" ต่างกันมาก

JavaScript ทำงานทีละอย่าง

ข้อเท็จจริงที่ต้องรู้ก่อนทุกอย่าง: JavaScript ในเบราว์เซอร์ทำงานบน thread เดียว คือทำได้ทีละอย่างเท่านั้น ไม่มีการทำสองอย่างพร้อมกัน และ thread เดียวกันนี้ก็คือ thread ที่วาดหน้าเว็บ ตอบสนองการคลิก และ scroll ด้วย พูดง่าย ๆ คือถ้าโค้ดของคุณยึด thread ไว้ หน้าเว็บก็ค้าง

ไม่ต้องเชื่อผม ลองเองเลย พิมพ์ชุดนี้ลง console แล้วรีบลองกดปุ่มหรือ scroll หน้าเว็บระหว่างที่มันทำงาน

// พิมพ์ลงใน console แล้วลองกดปุ่มหรือ scroll หน้าเว็บระหว่างที่มันทำงาน
console.log("เริ่ม");
const start = Date.now();
while (Date.now() - start < 3000) {
  // วนเปล่า ๆ 3 วินาที
}
console.log("จบ");

สามวินาทีนั้นหน้าเว็บตายสนิท คลิกไม่ได้ ปุ่มไม่ตอบสนอง อะไรบนหน้าจอก็ไม่ขยับ เพราะ loop ของเรากำลังกอด thread ไว้ไม่ปล่อย เบราว์เซอร์ไม่มีโอกาสได้ไปวาดอะไรเลย ทีนี้ลองนึกว่าถ้าการรอข้อมูลจากเซิร์ฟเวอร์ทำงานแบบนี้ ทุกครั้งที่แอปโหลดข้อมูล ผู้ใช้จะเจอหน้าค้างทุกครั้ง

ทีนี้ลองอันนี้บ้าง แต่ก่อนกด Enter ให้ทายก่อนว่าอะไรจะขึ้นมาก่อนหลัง

console.log("เริ่ม");
setTimeout(() => {
  console.log("ผ่านไป 3 วินาที");
}, 3000);
console.log("จบ");

ผลที่ได้คือ เริ่ม แล้ว จบ ขึ้นมาติดกันทันที จากนั้นอีกสามวินาทีค่อยมี ผ่านไป 3 วินาที โผล่ตามมา และที่สำคัญไม่แพ้กัน — ระหว่างสามวินาทีนั้นหน้าเว็บยังใช้งานได้ปกติทุกอย่าง คลิกได้ scroll ได้

สิ่งที่เกิดขึ้นคือ setTimeout ไม่ได้ "หยุดรอ" มันแค่ฝากงานไว้กับเบราว์เซอร์ว่า "อีกสามวินาทีช่วยเรียกฟังก์ชันนี้ให้ที" แล้วคืน thread ให้ทันที โค้ดบรรทัดถัดไปจึงวิ่งต่อได้เลย พอครบเวลาเบราว์เซอร์ค่อยเอาฟังก์ชันนั้นกลับมารันตอนที่ thread ว่าง

ช่องว่างระหว่าง "โค้ดที่สั่งให้เริ่มงาน" กับ "โค้ดที่จัดการผลลัพธ์" ตรงนี้แหละคือหัวใจของทั้งบท ทุกอย่างที่เหลือคือวิธีเขียนช่องว่างนี้ให้อ่านรู้เรื่อง

💡 ฟังก์ชันลูกศรที่เราส่งให้ setTimeout มีชื่อเรียกว่า callback — ฟังก์ชันที่เราไม่ได้เรียกเอง แต่ยื่นให้คนอื่นเก็บไว้เรียกให้ทีหลัง คุณใช้มันมาแล้วตั้งแต่คอร์ส JavaScript พื้นฐาน ทั้งใน addEventListener ("เมื่อมีคนคลิก ช่วยเรียกอันนี้") และใน map / filter ("ช่วยเรียกอันนี้กับสมาชิกทุกตัว") ที่นี่เป็นแนวคิดเดียวกันเป๊ะ เพียงแต่สิ่งที่รอคือเวลาแทนที่จะเป็นการคลิก

Promise: ใบเสร็จที่ยังรับของไม่ได้

callback ใช้ได้ดีกับงานเดี่ยว ๆ แต่พอเริ่มมีหลายขั้นตอนต่อกันมันจะเริ่มพัน ๆ JavaScript สมัยใหม่จึงมีวัตถุมาตรฐานตัวหนึ่งไว้แทน "ผลลัพธ์ที่ยังไม่พร้อม" ชื่อว่า Promise

วิธีคิดที่ง่ายที่สุดคือใบเสร็จร้านอาหาร คุณสั่งอาหารแล้วได้ใบเสร็จมาทันที ใบเสร็จนั้นไม่ใช่อาหาร กินไม่ได้ แต่มันคือคำสัญญาว่าเดี๋ยวอาหารจะมา ระหว่างนั้นคุณเอาใบเสร็จไปนั่งรอที่โต๊ะ คุยกับเพื่อน ทำอย่างอื่นได้ตามปกติ — นี่คือสิ่งที่ Promise เป็น มันคือของที่คุณได้รับทันที เพื่อแทนของที่ยังมาไม่ถึง

Promise หนึ่งตัวมีได้สามสถานะ และการเปลี่ยนสถานะเกิดขึ้นครั้งเดียว แล้วเปลี่ยนกลับไม่ได้อีกเลย

  • pending — ยังรออยู่ ยังไม่รู้ผล (อาหารยังทำไม่เสร็จ)
  • fulfilled — สำเร็จแล้ว และมีค่าผลลัพธ์ติดมาด้วย (อาหารมาเสิร์ฟแล้ว)
  • rejected — ล้มเหลว และมี error ติดมาด้วย (ครัวบอกว่าของหมด)

ปกติเราแทบไม่ต้องสร้าง Promise เองหรอก เพราะ fetch ในบทหน้าคืน Promise มาให้อยู่แล้ว แต่การลองสร้างสักตัวด้วยมือจะทำให้คุณเข้าใจว่าข้างในมันมีอะไร ลองอันนี้

const wait = (ms) => new Promise((resolve) => setTimeout(resolve, ms));

console.log(wait(2000)); // Promise { <pending> }

อ่านทีละชิ้น new Promise รับฟังก์ชันหนึ่งตัวเข้าไป และฟังก์ชันนั้นจะได้พารามิเตอร์มาสองตัวคือ resolve (เรียกเมื่องานสำเร็จ พร้อมส่งค่าผลลัพธ์ไปด้วยได้) กับ reject (เรียกเมื่องานพัง พร้อมส่ง error ไปด้วย) ในกรณีนี้เราไม่ได้ใช้ reject เลยเพราะการรอเวลาไม่มีทางพัง เราแค่บอกว่า "อีก ms มิลลิวินาที ช่วยเรียก resolve ให้ที" ซึ่งก็คือส่ง resolve เป็น callback ให้ setTimeout ตรง ๆ นั่นเอง

สังเกตบรรทัดที่สอง เรา console.log ผลของ wait(2000) ทันที และได้ Promise { <pending> } ออกมา ไม่ใช่ค่าอะไรที่ใช้ได้ — เพราะเราเพิ่งได้ใบเสร็จมา อาหารยังไม่มา ลองดูสถานะของมันเปลี่ยนกับตาก็ได้: กดลูกศรหน้าบรรทัดนั้นเพื่อกางออก คุณจะเห็นบรรทัด [[PromiseState]]: "pending" อยู่ข้างใน จากนั้นรอสักสองวินาที แล้วพับลูกศรกลับเข้าไปแล้วกางออกมาใหม่ คราวนี้มันจะกลายเป็น [[PromiseState]]: "fulfilled" (ต้องพับแล้วกางใหม่จริง ๆ เพราะข้อความที่ console แสดงไว้แล้วจะไม่อัปเดตตัวเองตามเวลา มันคำนวณค่าใหม่ตอนที่คุณกางออกมาดูเท่านั้น) แต่อย่ากด F5 โหลดหน้าใหม่เด็ดขาด เพราะ wait ที่เราเพิ่งประกาศไว้จะหายไปพร้อมกับหน้าเว็บ แล้วตัวอย่างที่เหลือทั้งบทจะฟ้อง ReferenceError: wait is not defined

📌 จำหน้าตาของ wait ตัวนี้ไว้ให้ดี เพราะคุณจะเจอมันอีกในบทที่ 13 ในรูป await new Promise((resolve) => setTimeout(resolve, 3000)) ที่ถูกใส่ไว้กลาง ๆ ฟังก์ชันโหลดข้อมูล นั่นคือการถ่วงเวลาปลอมที่ใส่ไว้ตั้งใจ เพื่อให้เราเห็นสถานะ Loading นาน ๆ พอที่จะสังเกตได้ ไม่ใช่โค้ดที่จำเป็นต่อการทำงานจริง (และในบทนั้นก็มีคอมเมนต์บอกให้ลบทิ้งทีหลัง) พอคุณรู้ว่ามันคืออะไร บรรทัดนั้นจะไม่น่ากลัวอีกต่อไป

ทีนี้ลองอีกตัวที่พังได้บ้าง เดี๋ยวเราจะใช้มันไปตลอดทั้งบทที่เหลือ พิมพ์เก็บไว้ใน console ได้เลย

function maybeFail() {
  return new Promise((resolve, reject) => {
    setTimeout(() => {
      if (Math.random() > 0.5) {
        resolve("สำเร็จ!");
      } else {
        reject(new Error("พังซะแล้ว"));
      }
    }, 1000);
  });
}

ตัวนี้รอหนึ่งวินาที แล้วโยนเหรียญ ถ้าออกหัวก็ resolve พร้อมข้อความ ถ้าออกก้อยก็ reject พร้อม Error มันคือแบบจำลองของ API request ในชีวิตจริงที่บางครั้งก็สำเร็จ บางครั้งก็ล่ม — และโค้ดของเราต้องรับมือได้ทั้งสองทาง

อ่านค่าจาก Promise ด้วย .then()

คำถามต่อไปคือแล้วเราจะเอาค่าออกจาก Promise ยังไง คำตอบคือเอาออกตรง ๆ ไม่ได้ ไม่มีทางเขียน const value = somePromise.value แล้วได้ของ สิ่งที่เราทำได้คือฝากไว้ล่วงหน้าว่าพอค่ามาถึงแล้วให้ทำอะไร ซึ่งก็คือหน้าที่ของเมธอด .then() และ .catch()

wait(2000).then(() => {
  console.log("ครบ 2 วินาทีแล้ว");
});

maybeFail()
  .then((message) => console.log("ได้ค่า:", message))
  .catch((error) => console.log("จับ error ได้:", error.message));

.then() คือ "ถ้าสำเร็จ ให้ทำอันนี้ โดยรับค่าที่ resolve ส่งมาเป็น argument" ส่วน .catch() คือ "ถ้าพัง ให้ทำอันนี้ โดยรับสิ่งที่ reject ส่งมาแทน" ลองรัน maybeFail() ชุดนี้สักห้าหกครั้ง คุณจะเห็นทั้งสองกิ่งสลับกันไปมา นั่นคือสิ่งที่เราต้องการ — โค้ดของเราครอบคลุมทั้งกรณีสำเร็จและกรณีพังในคราวเดียว

ปัญหาของ .then() จะเริ่มโผล่เมื่องานมีหลายขั้นและขั้นหลังต้องรอขั้นหน้าเสร็จก่อน ลองดูหน้าตาของงานสามขั้นที่ทำต่อกัน

wait(1000).then(() => {
  console.log("ขั้นที่ 1 เสร็จ");
  wait(1000).then(() => {
    console.log("ขั้นที่ 2 เสร็จ");
    wait(1000).then(() => {
      console.log("ขั้นที่ 3 เสร็จ");
    });
  });
});

เห็นโค้ดที่ค่อย ๆ ไถลไปทางขวาไหม แค่สามขั้นก็เริ่มอ่านยากแล้ว ถ้าเป็นห้าขั้นและมี error handling ปนอยู่ด้วยจะยิ่งกว่านี้มาก คนเรียกอาการนี้ว่า "callback hell" หรือ "pyramid of doom" (มันเขียนให้แบน ๆ ได้ด้วยการ return Promise ออกจาก .then() แล้วต่อ .then() ถัดไป แต่ก็ยังอ่านไม่ค่อยเป็นธรรมชาติอยู่ดี) เราจะไม่ลงลึกกว่านี้ เพราะเรื่องนี้มีไว้แค่เพื่อให้คุณเห็นว่าทำไมถึงต้องมีสิ่งที่จะพูดถึงต่อไปนี้

เขียนให้อ่านง่ายขึ้นด้วย async / await

async และ await คือคำสั่งสองคำที่ให้เราเขียนโค้ดแบบ asynchronous ด้วยหน้าตาของโค้ดธรรมดาบรรทัดต่อบรรทัด ไม่มีการไถลไปทางขวา ไม่มี callback ให้ต้องตามอ่าน ลองอันนี้

async function demo() {
  console.log("เริ่มรอ");
  await wait(2000);
  console.log("รอครบแล้ว");
}

demo();

อ่านแล้วเหมือนโค้ดปกติเลยใช่ไหม บรรทัดบนทำก่อน บรรทัดล่างทำทีหลัง await แปลว่า "หยุดตรงนี้จนกว่า Promise ตัวนี้จะเสร็จ แล้วค่อยไปบรรทัดถัดไป โดยเอาค่าที่ resolve ส่งมาเป็นผลของนิพจน์นี้" ถ้าจะเขียนใหม่ให้ครบก็คือ const value = await somePromise; — ตรงไปตรงมาเหมือนกำหนดค่าตัวแปรทั่วไป

มีสามเรื่องเกี่ยวกับสองคำนี้ที่ต้องเข้าใจให้ชัด ไม่งั้นจะสับสนแน่นอน

หนึ่ง: await ใช้ได้เฉพาะภายในฟังก์ชันที่เป็น async เท่านั้น ถ้าคุณเขียน await ในฟังก์ชันธรรมดา JavaScript จะฟ้อง syntax error ทันที คิดง่าย ๆ ว่า async คือการติดป้ายหน้าฟังก์ชันว่า "ฟังก์ชันนี้อาจจะหยุดรอกลางทางนะ" ซึ่งเป็นข้อมูลที่ทั้งคนอ่านและ JavaScript engine ต้องรู้ล่วงหน้า

🔍 console ของ DevTools เป็นข้อยกเว้นพิเศษ มันยอมให้คุณพิมพ์ await ที่ระดับบนสุดได้เลยโดยไม่ต้องห่อ async function (เรียกว่า top-level await) นั่นคือเหตุผลที่ตัวอย่างในบทที่ 12 พิมพ์ลง console ตรง ๆ แล้วทำงานได้ แต่ในไฟล์โค้ดจริงของโปรเจกต์ React ของเรา กฎข้อนี้ยังบังคับใช้เต็มที่ อย่าเผลอเอาความเคยชินจาก console ไปใช้ในไฟล์

สอง: await หยุดแค่ฟังก์ชันตัวนั้น ไม่ได้หยุดทั้งหน้าเว็บ ข้อนี้เข้าใจผิดกันบ่อยที่สุด คำว่า "หยุดรอ" ทำให้หลายคนนึกถึง while loop สามวินาทีที่เราลองกันตอนต้นบท — ไม่ใช่เลย มันคนละเรื่องกันโดยสิ้นเชิง ระหว่างที่ demo() ค้างอยู่ที่บรรทัด await thread ถูกปล่อยคืนให้เบราว์เซอร์ไปทำอย่างอื่นเรียบร้อยแล้ว ปุ่มยังกดได้ animation ยังวิ่ง โค้ดส่วนอื่นยังทำงาน แล้วพอ Promise เสร็จ JavaScript ค่อยกลับมาทำบรรทัดถัดไปของ demo() ต่อจากจุดเดิม ลองพิสูจน์ดูก็ได้ ระหว่างที่ demo() กำลังรอ ให้พิมพ์ console.log("ฉันยังทำงานได้") ลงไปแล้วกด Enter — มันขึ้นมาทันที ไม่ต้องรอ

สาม: ฟังก์ชัน async คืน Promise เสมอ ไม่ว่าข้างในจะ return อะไรออกมาก็ตาม แม้แต่ตัวเลขเปล่า ๆ

async function giveNumber() {
  return 42;
}

console.log(giveNumber()); // Promise { 42 } — ไม่ใช่ 42
console.log(await giveNumber()); // 42

ติดคำว่า async ไว้หน้าฟังก์ชันเมื่อไร ค่าที่ return จะถูกห่อด้วย Promise ให้อัตโนมัติเสมอ ถ้าอยากได้ค่าข้างในก็ต้อง await ออกมา ฟังดูเป็นรายละเอียดจุกจิก แต่จำข้อนี้ไว้ให้ดี เพราะมันคือเหตุผลตรง ๆ ของกฎข้อหนึ่งที่คุณจะเจอตอนท้ายบทนี้

จัดการ error ด้วย try / catch

พอเลิกใช้ .then() แล้ว .catch() ก็หายไปด้วย คำถามคือแล้วเราจะดัก error ยังไง คำตอบคือใช้ try / catch ซึ่งเป็นไวยากรณ์ดักข้อผิดพลาดแบบปกติของ JavaScript นั่นแหละ นี่คือข้อดีที่ถูกมองข้ามของ await — มันทำให้โค้ด asynchronous กลับมาใช้เครื่องมือชุดเดียวกับโค้ดธรรมดาได้

async function run() {
  try {
    const message = await maybeFail();
    console.log("ได้ค่า:", message);
  } catch (error) {
    console.log("จับ error ได้:", error.message);
  } finally {
    console.log("ทำงานเสมอ ไม่ว่าจะสำเร็จหรือพัง");
  }
}

run();

อ่านจากบนลงล่างได้ตรง ๆ เลย: ลองทำสิ่งที่อยู่ใน try ถ้าระหว่างนั้นมี Promise ตัวไหน reject ให้กระโดดไปที่ catch ทันทีพร้อมกับ error ตัวนั้น และไม่ว่าจะจบทางไหน บล็อก finally จะได้ทำงานเสมอ รัน run() ซ้ำหลาย ๆ ครั้งเพื่อดูทั้งสองทาง

ทีนี้ลองดูว่าถ้าลืม try / catch จะเป็นยังไง อันนี้ต้องเห็นกับตาสักครั้ง

async function runWithoutTry() {
  const message = await maybeFail(); // ไม่มี try/catch ครอบไว้
  console.log("ได้ค่า:", message);
}

runWithoutTry();

รันไปเรื่อย ๆ จนกว่าเหรียญจะออกก้อย แล้วคุณจะเห็นข้อความสีแดงใน console ว่า Uncaught (in promise) Error: พังซะแล้ว สังเกตคำว่า in promise ให้ดี — นี่คือ error ที่ไม่มีใครรับ มันหลุดออกจากฟังก์ชันไปตายอยู่กลางอากาศ ในแอปจริงอาการที่ผู้ใช้เห็นคือหน้าจอค้างอยู่ที่ Loading ตลอดกาลโดยไม่มีอะไรบอกว่าเกิดอะไรขึ้น ซึ่งแย่กว่าการขึ้นข้อความ error เสียอีก

⚠️ ทุกครั้งที่คุณเขียน await ให้ถามตัวเองว่า "ถ้าอันนี้พังขึ้นมา ใครรับ" การขอข้อมูลจากเน็ตเวิร์กเป็นสิ่งที่พังได้เป็นปกติ ไม่ใช่กรณีพิเศษ — เน็ตหลุด เซิร์ฟเวอร์รีสตาร์ต ผู้ใช้เข้าลิฟต์ ทั้งหมดนี้เกิดขึ้นจริงทุกวัน

ส่วน finally อาจดูเหมือนของแถมที่ไม่ค่อยได้ใช้ แต่จริง ๆ มันมีที่ทางที่ชัดเจนมากในงาน UI: มันคือที่สำหรับปิดสถานะ loading เพราะไม่ว่าข้อมูลจะมาถึงหรือ request จะล่ม สิ่งหนึ่งที่จริงเสมอคือ "เราไม่ได้กำลังโหลดอยู่แล้ว" การเขียน setLoading(false) ไว้ใน finally จึงทำให้ไม่มีทางลืมกรณีใดกรณีหนึ่ง — และนั่นคือ state ตัวเดียวกับที่บทที่ 13 จะสร้างขึ้นมาคุมข้อความ Loading บนหน้าจอ

เกร็ดที่จะได้ใช้ในบทที่ 13

⚠️ ก่อนจบบท มีกฎข้อหนึ่งที่อยากให้ติดตัวไว้ล่วงหน้า เพราะมันเป็นจุดที่คนสะดุดกันแทบทุกคน: คุณส่งฟังก์ชัน async ให้ useEffect ตรง ๆ ไม่ได้ เหตุผลคือ useEffect คาดหวังว่า callback ที่มันได้รับจะคืนค่าว่าง หรือไม่ก็คืนฟังก์ชัน cleanup กลับมาเท่านั้น ส่วนฟังก์ชัน async คืน Promise เสมอ อย่างที่เราเพิ่งพิสูจน์กันไปกับ giveNumber() — React จึงได้ Promise มาแทนที่จะเป็นฟังก์ชัน cleanup ที่มันต้องการ ทางออกไม่ได้ซับซ้อนอะไรเลย แค่ประกาศฟังก์ชัน async แยกไว้ต่างหาก แล้วเรียกมันจากข้างใน effect

// รูปแบบที่จะเจอในบทที่ 13
useEffect(() => {
  fetchSomething(); // เรียก async function ที่ประกาศแยกไว้
}, []);

callback ที่ส่งให้ useEffect ยังเป็นฟังก์ชันธรรมดาอยู่ ไม่มี async ไม่มี await คืนค่าว่าง ๆ ตามที่ React ต้องการ ส่วนงานที่ต้องรอทั้งหมดไปอยู่ในฟังก์ชันแยกที่เราเรียกออกไปแล้วปล่อยให้มันทำงานของมันไป

ตอนนี้คุณมีครบแล้วทั้ง Promise, async, await และ try / catch — และรู้แล้วว่าแต่ละอันแก้ปัญหาอะไร ในบทที่ 12 เราจะเอาทั้งชุดนี้ไปใช้กับของจริง คือยิง HTTP request ไปหาเซิร์ฟเวอร์ด้วย fetch แล้วอ่านข้อมูลที่มันส่งกลับมา จากนั้นในบทที่ 13 ทุกอย่างจะมารวมกันใน React
